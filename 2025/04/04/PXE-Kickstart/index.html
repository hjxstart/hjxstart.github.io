# linux加载内核文件，initrd加载虚拟根文件；
# linux和linuxefi是一样的，initrd和initrdefi是一样的
# 内核文件/var/lib/tftpboot/kernel/{Kylin}
# R系列；/images/pxeboot/目录下的vmlinuz 和 initrd.img，镜像需要挂载在目录上。
menuentry --hotkey=c "c  Centos7.9" {
        set gfxpayload=keep
        echo "Loading Centos7.9 Kernel..."
        linux kernel/Centos7.9/vmlinuz inst.repo="http://${IP}/iso/Centos7.9" inst.ks="http://${IP}/ks/Centos7.9.cfg"
        echo "Loading Centos7.9 RamDisk..."
        initrd kernel/Centos7.9/initrd.img
}

        set gfxpayload=keep

	linux kernel/Ubuntu2204/vmlinuz ip=dhcp url="http://${IP}/iso/src/Ubuntu2204.iso"
        echo "Loading Ubuntu2204 RamDisk..."
        initrd kernel/Ubuntu2204/initrd.img
}

#!/bin/bash

echo "请输入bond接口名称"
read bond_name

echo "请输入bond ip（如192.168.1.1/24）"
read bond_ip

echo "请输入bond 网关（如192.168.1.254）"
read bond_gateway

echo "请输入物理网卡名称（如 eth0,eth1）"
read physical_interfaces

#将物理网卡列表转为数组
IFS=',' read -ra physical_interfaces_array <<< "$physical_interfaces"

bond_create() {
    #创建bond 接口
    nmcli connection add type bond con-name $bond_name ifname $bond_name mode $bond_mode
    if [ $? -ne 0 ]; then
        echo "bond 接口 $bond_name 创建失败"
        exit 1
    fi
    #将物理网卡添加到bond接口
    for interface in "${physical_interfaces_array[@]}"; do
        nmcli connection add type ethernet slave-type bond con-name "${interface}-slave" ifname $interface master $bond_name
        if [ $? -ne 0 ]; then
            echo "将网卡$interface 添加到bond接口失败"
            exit 1
        fi
    done
    #配置bond接口的ip
    nmcli connection modify $bond_name ipv4.addresses $bond_ip  ipv4.gateway $bond_gateway  ipv4.method manual
    if [ $? -ne 0 ]; then
        echo "修改 $bond_name 网络参数失败"
        exit 1
    fi

    #激活bond 接口
    nmcli connection up $bond_name
    if [ $? -ne 0 ]; then
        echo "$bond_name 激活失败"
        exit 1
    fi
    echo "$bond_name 成功配置及激活"

}

echo "bond配置菜单如下："
echo "1.配置bond0（balance-rr）轮询"
echo "2.配置bond1（active-backup）主备"
echo "3.配置bond4（802.3ad）动态聚合链路"

read Bond

case $Bond in
    1)
        bond_mode="balance-rr"
        bond_create
        ;;
    2)
        bond_mode="active-backup"
        bond_create
        ;;
    3)
        bond_mode="802.3ad"
        bond_create
        ;;
    *)
        echo "请输入1-3数字"
        ;;
esac

auto_detect.sh

#!/bin/bash

# =========================================================
#        麒麟操作系统 (Kylin V10) 自动化交付系统 V2.9
# =========================================================

# --- 核心配置区 ---
NETWORK="192.168.215.0/24"       #
EXCLUDE_IP="192.168.215.2"       #
SSH_PASS="服务器的密码【待按需修改----------------------------------】"           #
SN_DONE_FILE="finished_sn.txt"   #
LOCK_DIR="/tmp/pxe_locks"        #
MAX_CONCURRENT=10                #
HOST_MAP="host_map.yml"          #
ERROR_LOG="/tmp/ansible_error.log"

# 终端颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m' # 无颜色

# --- 1. 环境自检 ---
echo -e "${BLUE}[系统] 正在初始化环境...${NC}"
mkdir -p $LOCK_DIR
touch $SN_DONE_FILE

for tool in nmap sshpass nc ansible-playbook; do
    if ! command -v $tool &> /dev/null; then
        echo -e "${RED}[错误] 未安装必要工具: $tool，请先安装！${NC}"
        exit 1
    fi
done

if [ ! -f "$HOST_MAP" ]; then
    echo -e "${RED}[错误] 找不到资产映射表: $HOST_MAP${NC}"
    exit 1
fi

shopt -s nocasematch #

echo -e "${GREEN}========================================================="
echo -e "        麒麟自动化交付系统 V2.9 - 调试模式已开启"
echo -e "=========================================================${NC}"

while true; do
    RUNNING=$(ls $LOCK_DIR 2>/dev/null | wc -l)
    DONE=$(wc -l < $SN_DONE_FILE)

    # 2. 扫描阶段提示
    echo -e "${BLUE}[$(date +%H:%M:%S)] 正在扫描网段 $NETWORK ... (并发中: $RUNNING | 已完成: $DONE)${NC}"
    IPS=$(nmap -n -F --open $NETWORK -oG - | awk '/Up/{print $2}' | grep -vE "$EXCLUDE_IP")

    if [ -z "$IPS" ]; then
        echo -e "${YELLOW}[提示] 网段内未发现任何存活主机，10秒后重新扫描...${NC}"
    fi

    for IP in $IPS; do
        if [ $(ls $LOCK_DIR 2>/dev/null | wc -l) -ge $MAX_CONCURRENT ]; then
            echo -e "${YELLOW}[限制] 达到最大并发数 $MAX_CONCURRENT，等待中...${NC}"
            break
        fi

        # 3. 探测阶段提示
        echo -e "[监控] 发现主机: $IP，正在测试 SSH (22端口)..."
        if ! nc -z -w 2 $IP 22 &>/dev/null; then
            echo -e "${RED}[跳过] $IP: SSH 端口未开放${NC}"
            continue
        fi

        # 4. SN 获取阶段提示
        echo -e "[监控] $IP: SSH 已就绪，尝试提取序列号..."
        ID=$(sshpass -p "$SSH_PASS" ssh -o StrictHostKeyChecking=no -o ConnectTimeout=2 root@$IP "dmidecode -s system-serial-number 2>/dev/null" | tr -d '[:space:]' | tr -d '\r')

        if [ -z "$ID" ]; then
            echo -e "[监控] $IP: dmidecode 提取失败，尝试 UUID..."
            ID=$(sshpass -p "$SSH_PASS" ssh -o StrictHostKeyChecking=no root@$IP "cat /sys/class/dmi/id/product_uuid 2>/dev/null" | cut -c1-8)
        fi

        if [ -z "$ID" ]; then
            echo -e "${RED}[警告] $IP: 无法获取任何硬件标识符，放弃该主机。${NC}"
            continue
        fi

        # 5. 匹配逻辑提示
        echo -e "[监控] $IP: 提取到 ID: ${YELLOW}$ID${NC}，正在对比 $HOST_MAP ..."

        # 兼容 pattern_configs 里的 sn_regex 提取
        MAP_KEYS=$(grep -oP '(sn_regex:\s*"\K[^"]+)|(^\s*"\K[^"]+(?=":))' "$HOST_MAP")
        MATCHED=false

        for KEY in $MAP_KEYS; do
            if [[ "$ID" =~ $KEY ]]; then
                MATCHED=true
                break
            fi
        done

        if [ "$MATCHED" = false ]; then
            echo -e "${YELLOW}[跳过] ID: $ID 不在资产映射表中。${NC}"
            continue
        fi

        if grep -qi "$ID" "$SN_DONE_FILE"; then
            echo -e "${BLUE}[忽略] ID: $ID 之前已处理过。${NC}"
            continue
        fi

        if [ -f "$LOCK_DIR/$ID" ]; then
            echo -e "${YELLOW}[冲突] ID: $ID 正在被另一个线程处理。${NC}"
            continue
        fi

        # 6. 执行阶段提示
        touch "$LOCK_DIR/$ID"
        echo -e "${GREEN}[核心] >>> 匹配成功！准备为 $IP (SN: $ID) 执行 Ansible Playbook...${NC}"

        {
            #
            ansible-playbook deploy-server.yml -i "$IP," \
                -e "ansible_user=root ansible_password=$SSH_PASS" \
                --ssh-common-args='-o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -o LogLevel=ERROR' \
                >> "$ERROR_LOG" 2>&1

            if [ $? -eq 0 ]; then
                echo "$ID | $IP | $(date)" >> "$SN_DONE_FILE"
                echo -e "${GREEN}[成功] <<< $ID ($IP) 部署任务已完成。${NC}"
            else
                echo -e "${RED}[失败] <<< $ID 执行报错，请查看日志: tail -n 20 $ERROR_LOG${NC}"
            fi
            rm -f "$LOCK_DIR/$ID"
        } &
    done

    sleep 10
done

deploy-server.yml

---
- name: "PXE 专业硬件深度巡检"
  hosts: all
  gather_facts: yes
  vars_files:
    - host_map.yml
  vars:
    report_temp_dir: "./temp_reports"

  tasks:
    # 设置主机名
    # --- 1.1 提取并深度清洗 SN ---
    - name: "1.1 提取标准化 SN"
      set_fact:
        # 强制删除空格、换行、反引号等非法字符，并转大写
        norm_sn: "{{ ansible_product_serial | regex_replace('[^a-zA-Z0-9]', '') | upper }}"

    # --- 1.2 精确匹配 (优先级 1) ---
    - name: "1.2.1 尝试精确匹配 (unique_configs)"
      set_fact:
        cfg: "{{ unique_configs[item] }}"
      loop: "{{ unique_configs.keys() | list }}"
      when:
        - cfg is not defined
        # 将配置文件里的 SN 键名也转大写进行对比，实现大小写无关匹配
        - norm_sn == (item | regex_replace('[^a-zA-Z0-9]', '') | upper)

    # --- 1.2.2 正则匹配 (优先级 2) ---
    - name: "1.2.2 尝试正则匹配 (pattern_configs)"
      set_fact:
        cfg: "{{ item }}"
      loop: "{{ pattern_configs }}"
      when:
        - cfg is not defined
        - norm_sn is search(item.sn_regex, ignorecase=True)
      no_log: true

    # --- 1.3 确定最终主机名逻辑 ---
    - name: "1.3 逻辑处理：确定最终主机名"
      set_fact:
        # 核心逻辑：有 hostname 用 hostname；没有则用 prefix + SN 后四位
        raw_hostname: "{{ cfg.hostname if cfg.hostname is defined else (cfg.hostname_prefix + '-' + norm_sn[-4:]) }}"
      when: cfg is defined

    - name: "1.3.1 强制清洗主机名 (防止 Invalid hostname 报错)"
      set_fact:
        # 再次过滤，确保只含字母、数字和中横杠，转小写
        final_hostname: "{{ raw_hostname | regex_replace('[^a-zA-Z0-9-]', '') | lower }}"
      when: cfg is defined
      
    - name: "1.4 执行修改系统主机名"
      hostname:
        name: "{{ final_hostname }}"
      when: cfg is defined


    # 配置网络 Bond4
    # 2.1 先清理旧连接并创建 Bond 主接口 (仅执行一次)
    - name: "2.1 初始化 Bond4 主接口"
      shell: |
        nmcli conn delete '{{ cfg.bond_name | default("bond0") }}' || true
        nmcli conn add type bond con-name {{ cfg.bond_name | default('bond0') }} ifname {{ cfg.bond_name | default('bond0') }} mode 4
        nmcli conn modify {{ cfg.bond_name | default('bond0') }} ipv4.addresses {{ cfg.ip_address }}/24 ipv4.gateway {{ cfg.gateway }} ipv4.method manual
      when: cfg is defined

    # 2.2 循环添加从属网卡 (Slave)
    - name: "2.2 绑定物理网卡到 Bond"
      shell: |
        nmcli conn delete '{{ item }}' || true
        nmcli conn add type bond-slave con-name {{ cfg.bond_name | default('bond0') }}-{{ item }} ifname {{ item }} master {{ cfg.bond_name | default('bond0') }}
      loop: "{{ cfg.bond_slaves }}"
      when: cfg is defined

    # 2.3 统一启动
    - name: "2.3 启动 Bond 网络"
      shell: "nmcli conn up {{ cfg.bond_name | default('bond0') }}"
      when: cfg is defined


    ## 收集环境信息
    - name: "1. 采集设备 SN"
      shell: "dmidecode -s system-serial-number | tr -d ' '"
      register: sn_cmd

    - name: "4. 内存资产采集"
      # 采集插槽数量及大小
      shell: |
        dmidecode -t memory | grep -E "Size:|Serial Number:" | grep -v "No Module Installed" | sed 's/^[ \t]*//g'
      register: mem_details
      
    - name: "5. 硬盘资产采集"
      # 采集名称、型号、大小及物理 SN
      shell: "lsblk -dno NAME,MODEL,SIZE,SERIAL,WWN | grep -v 'loop'"
      register: disk_details

    # --- 修正网络资产采集 ---
    - name: "6. 网络资产采集 (接口/IP/MAC 一体化)"
      # 改进逻辑：逐行扫描接口，并将接口名、IP、MAC 放在同一行显示
      shell: |
        echo "--- 物理网卡硬件 ---"
        lspci | grep -i net
        echo -e "\n--- 网络接口详细映射 ---"
        # 提取接口名、MAC、以及对应的 IPv4 地址
        ip -br link show | awk '{print $1}' | while read iface; do
          MAC=$(ip link show $iface | awk '/link\/ether/ {print $2}')
          IP=$(ip -4 addr show $iface | awk '/inet / {print $2}' | paste -sd "," -)
          printf "接口: %-10s | MAC: %-17s | IP: %s\n" "$iface" "${MAC:-N/A}" "${IP:-未分配}"
        done
      register: nic_details

    - name: "6.1 采集 Bond 绑定状态"
      # 从 /proc/net/bonding/ 获取详细的从属网卡和链路状态
      shell: |
        if [ -d "/proc/net/bonding" ]; then
          for bond in /proc/net/bonding/*; do
            echo "--- Bond 接口: ${bond##*/} ---"
            grep -E "Bonding Mode|Slave Interface|MII Status" $bond

2. PXE服务端初始化

## PXE和KickStart
首先要明确：两者是不同的技术，起着不同的作用
# PXE预启动执行环境（Pre-boot Execution Environment）
1. 也称为预执行环境，作用：通过网络启动操作系统的安装程序
2. 是Intel公司研发的一种网络引导协议，只要是适用Inter的CPU的服务器，器BIOS都支持PXE技术。支持客户端通过网络从服务器下载启动程序及映像等，并由此通过网络启动操作系统。基于Client/Server的网络模式（即客户端与服务器），所以需要配置PXE服务端和PXE客户端。
# PXE服务端
1. PXE服务端可以部署在多种不同的Linux发行版本上，至少由三种服务组成：DHCP、TFTP、HTTP/FTP/NFS。
2. DHCP：负责给客户端分配IP地址、定位PXE启动引导文件以及指定TFTP服务器的IP；
3. TFTP（trivial file transfer protocol）：负责将PXE引导程序、内核文件vmlinuz、临时根文件镜像initrd等文件传输给客户机。
4. HTTP/FTP/NFS：负责将安装操作系统所需的大文件传输给客户机
5. 这三种服务可以同时搭建在同一台服务器上，也可以分别搭建在不同的服务器上；还可以分别部署在不同的平台上，如DHCP服务由Windows系统提供，TFTP搭建在Linux平台，HTTP由Unix系统提供服务；甚至分别可以由不同的设备提供，如路由器提供DHCP服务，手机提供TFTP服务，服务器提供HTTP服务。

# TFTP介绍
Trivial File Transfer Protocol (简单文件传输协议，是文件传输协议FTP的简化版本)， 作用是：使用UDP协议在客户端和服务端之间进行简单文件传输，适合小文件传输的场景。
# TFTP特点：
1. TFTP是一种开放协议，缺乏安全性，没有加密机制，TFTP客户端与TFTP服务端通信时不需要身份认证
2. TFTP使用UDP（性能好，没有三次握手，可靠性低）作为传输层协议，使用69端口
3. TFTP只有5个指令可以执行（rrg, wrq, data, ack, error）
4. 基于C/S模式，所以可以在PXE服务端和客户端之间传输文件，CentOS7中，tftp-server: 服务端包；tftp: 客户端包
5. 网卡存储空间有限，所以只能集成轻量级别的ftp客户端。tftp客户端只有50多k，容量小，所以可以集成在PXE网卡芯片中
6. TFTP服务默认由xinetd服务进行管理，用xinetd重启tftp-server来使配置生效。# 修改tftp配置文件，需要重启xinetd服务使其生效

## Client需要的启动引导文件
1. 由于物理服务器硬盘基本都大于2.2TB，所以肯定是gpt分区而非mbr分区，因为mbr分区无法管理2.2T以上的空间。mbr分区一般和Legacy BIOS固件搭配，对于传统Legacy BIOS，PXE引导时使用(syslinux提供的) 一个名为pxelinux.0的启动文件来引导。
2. 所以只要看到一个pxe教程中使用的启动文件是pxelinux.0，那一定是用的Legacy BIOS固件，那么不用往下看了，这过时了，因为真实的物理服务器是不适用的。
3. 而gpt分区一般和EFI BIOS固件搭配，对于EFI BIOS,需要使用efi文件引导，为实现UEFI SecureBoot，大多数Linux使用shim.efi或bootx64.efi（经测试，文件名不限制，单必须以.efi结尾，client引导时会提示：Fetching Netboot Image，即获取Netboot映像）嵌套调用grubx64.efi（经测试，文件名不限制，但必须以.efi结尾）来引导。接着，grubx64.efi会加载同目录下的grub.cfg菜单文件，然后就可以选择安装哪个操作系统了。
4. 所以现在我们要准备至少3个文件：实现UEFI SecureBoot的shim.efi或boox64.efi；启动文件 grubx64.efi；菜单文件grub.cfg

## 什么是GNU
GNU的全程是 "GNU's Not Unix!", 它是GNU的递归缩写，它的正确发音是：葛奴，GNU项目由理查德·马修·斯托曼 (Richard Matthew Stallman) 发起。因为Unix是商业操作系统，所以在Richard看来是“不自由”的，正因此，Richard才发起了GNU项目。GNU==是一个类Unix操作系统（项目）==，GNU项目包括：操作系统内核（Linux只是内核之一、hurd）、编译器、调试器、开发工具、应用软件等自由软件。
## 什么是GRUB？
1. grub (Grand Unified Bootloader: 大一统引导程序)是GNU项目的一部分，是一个最常用的系统引导程序（管理器）--它负责将操作系统的内核加载到内存并执行它，一旦把内核加载，系统引导管理器的任务就算完成，系统引导的其他部分，比如操作系统的初始化及启动过程完全由内核来完成（系统引导管理器不直接加载操作系统）。GRUB可以引导多种操作系统，如：Linux/FreeBSD/OpenBSD/DOS和Windows系列等。
2. 现在常用的版本是grub2，grub2增添了许多语法，更接近脚本语言，例如有自己的命令、变量（系统变量、自定义变量）、条件判断、循环
## gfxterm

## 自动应答文件
之前已经实现了PXE的工作，接下来就可以一步一步的完成操作系统的手工安装（通常，在常规安装操作系统的过程中）。这就需要和服务器进行多次人工交互操作，比如选择语言、设置分区、设置网络等，为了减少手工安装的交互过程，可以事先定义好一个自动应答文件并存放在PXE服务端，该文件记录了安装过程中需要人工操作的各种参数（即：选择语言、设置分区、设置网络等这些），在以后通过PXE安装操作系统时，Clicent向服务端请求此文件后，Client的安装程序就可以从该文件中读取安装进行安装（当没有找到合适的参数时，则安装需要手动干预才能完成），这样就避免了在安装过程中多次的人机交互，无需人工干预，从而实现操作系统的自动化安装。自动应答文件甚至提供了一些常规安装无法实现的功能，比如添加epel源、添加别名等功能。
# 简单来说：
1. PXE的工作是通过网络为PXE Client启动安装程序，自动应答文件的作用是：实现操作系统的自动化安装，是一种无人值守安装方式。
2. PXE+自动应答文件结合麒麟使用时，就可以实现大规模的自动化操作系统部署。通过将两者结合起来，可以按如下方式自动安装：
打开PXE客户端电源 -> PXE在网络上启动安装程序 -> KickStart 自动进行安装 -> 完成全部安装任务后停留在启动就绪状态

### ks文件可用使用多种脚本 %post --log=/root/ks-post.log [post部分如果有问题就会记录]
比如：%pre预安装脚本和%post安装后脚本。在这两种脚本中都可以使用脚本语言（比如python，默认Bash）。%pre部分在OS安装开始之前、KS文件解析之后执行，所以使用场景有限，较少使用；而%post部分是在OS安装完成后、在第一次启动前执行，所以使用较多。这些脚本必须以%end结束，并且在ks文件中允许存在多个%pre和%post。%pre和%post部分的顺序不重要。
## %post安装后脚本
1. 当系统安装完成后、在第一次启动前，Kickstart会依次执行%post部分中的命令和脚本。这些脚本和命令可用用于进行各种系统定制操作，如安装软件包、配置网络、修改系统文件等。这意味着，用户可用通过%post部分来实现对系统进一步的定制，以满足特定需求。
2. 安装过程，实际上是先引导了内存中的: vmlinuz + initrd，以此为基础将需要的文件安装到目标系统（PXE客户端），此时安装介质被挂载到/run/install/repo, 目标系统被挂载到/mnt/sysimage。在%post阶段时，允许指定在内存系统中操作还是chroot（Change Root，也就是改变程序执行时所参考的根目录位置）到目标系统中操作，默认为chroot环境，即新安装好的系统，根目录其实是在/mnt/sysimage。由于安装后脚本在chroot环境中运行，因此大多数systemctl命令将拒绝执行任何操作。请注意，在执行%post部分的过程中，仍然必须插入安装介质。此部分对于安装其他软件或配置其他名称服务器等功能非常有用。
3. 如果配置了网络，当安装执行%post部分时，/etc/resolv.conf文件尚未完成。可以正常访问网络，但无法解析IP地址。因此，如果使用DHCP，必须在%post部分指定IP地址。如果为网络配置了静态IP西悉尼、包括名称服务器（DNS解析），则可以解析域名。
4. 如需要在目标系统上执行操作比如创建账号，那就以chroot模式运行。如
%post
touch /root/test.file # 安装完成后新系统的root下面有test.file文件。
5. 安装后脚本运行在chroot环境下，默认不支持从安装介质中拷贝脚本、RPM包等任务。您可以使用--nochroot选项更改此行行为，即：可以使用%post --nochroot来执行一些非chroot环境下的命令。这样，%post脚本将不再安装的目标系统中的chroot中运行。比如在chroot环境下是无法访问光驱的安装光盘的，因为光盘挂载在/tmp/cdrom，它是处于non-chroot环境下的。如果需要拷贝安装介质上的文件，就以--nochroot模式运行。
6. 一般来说post install脚本以#!/bin/sh这样的自定义shell作为第一行，然后下边就是普通的shell脚本了。
